import java.time.LocalDate; import java.time.temporal.ChronoUnit; import java.util.*; import java.util.stream.Collectors; /** * 图书馆管理系统 * 功能：图书管理、读者管理、借阅管理、归还管理、逾期计算、数据统计 */ public class LibraryManagementSystem { private final Map books = new HashMap<>(); private final Map readers = new HashMap<>(); private final Map borrowRecords = new HashMap<>(); private static final int MAX_BORROW_COUNT = 5; private static final int BORROW_DAYS = 30; private static final double DAILY_FINE = 0.5; public static void main(String[] args) { LibraryManagementSystem system = new LibraryManagementSystem(); system.initializeData(); system.displayAllBooks(); system.borrowBook("R001", "B001"); system.borrowBook("R002", "B002"); system.displayReaderBorrowingRecords("R001"); system.returnBook("R001", "B001"); system.displayStatistics(); } /** * 初始化测试数据 */ public void initializeData() { addBook(new Book("B001", "Java核心技术", "Cay S. Horstmann", 10)); addBook(new Book("B002", "深入理解Java虚拟机", "周志明", 8)); addBook(new Book("B003", "Effective Java", "Joshua Bloch", 6)); addBook(new Book("B004", "算法导论", "Thomas H. Cormen", 5)); addBook(new Book("B005", "设计模式", "Erich Gamma", 7)); addReader(new Reader("R001", "张三", "13800000001")); addReader(new Reader("R002", "李四", "13800000002")); addReader(new Reader("R003", "王五", "13800000003")); System.out.println("图书馆初始化完成。"); } /** * 添加图书 */ public void addBook(Book book) { Objects.requireNonNull(book, "图书不能为空"); if (books.containsKey(book.getId())) { throw new IllegalArgumentException("图书编号已存在：" + book.getId()); } books.put(book.getId(), book); System.out.println("新增图书成功：" + book.getTitle()); } /** * 借阅图书 */ public void borrowBook(String readerId, String bookId) { Reader reader = readers.get(readerId); Book book = books.get(bookId); if (reader == null) { System.out.println("读者不存在：" + readerId); return; } if (book == null) { System.out.println("图书不存在：" + bookId); return; } if (reader.getBorrowedBookIds().size() >= MAX_BORROW_COUNT) { System.out.println("已达到最大借阅数量。"); return; } if (book.getAvailableCount() <= 0) { System.out.println("图书库存不足：" + book.getTitle()); return; } if (reader.getBorrowedBookIds().contains(bookId)) { System.out.println("请勿重复借阅同一本图书。"); return; } LocalDate borrowDate = LocalDate.now(); LocalDate dueDate = borrowDate.plusDays(BORROW_DAYS); BorrowRecord record = new BorrowRecord( UUID.randomUUID().toString(), readerId, bookId, borrowDate, dueDate ); book.setAvailableCount(book.getAvailableCount() - 1); reader.getBorrowedBookIds().add(bookId); borrowRecords.put(record.getRecordId(), record); System.out.println("借阅成功：" + book.getTitle()); System.out.println("应还日期：" + dueDate); } /** * 归还图书 */ public void returnBook(String readerId, String bookId) { Reader reader = readers.get(readerId); Book book = books.get(bookId); if (reader == null || book == null) { System.out.println("读者或图书不存在。"); return; } if (!reader.getBorrowedBookIds().contains(bookId)) { System.out.println("该读者没有借阅此图书。"); return; } BorrowRecord record = borrowRecords.values() .stream() .filter(r -> r.getReaderId().equals(readerId) && r.getBookId().equals(bookId) && !r.isReturned()) .findFirst() .orElse(null); if (record == null) { System.out.println("未找到有效借阅记录。"); return; } LocalDate returnDate = LocalDate.now(); record.setReturnDate(returnDate); record.setReturned(true); long overdueDays = Math.max( 0, ChronoUnit.DAYS.between(record.getDueDate(), returnDate) ); double fine = overdueDays * DAILY_FINE; book.setAvailableCount(book.getAvailableCount() + 1); reader.getBorrowedBookIds().remove(bookId); System.out.println("归还成功：" + book.getTitle()); System.out.println("逾期天数：" + overdueDays); System.out.println("应缴罚款：" + fine + " 元"); } /** * 按书名搜索图书 */ public List searchBooks(String keyword) { if (keyword == null || keyword.trim().isEmpty()) { return new ArrayList<>(books.values()); } String normalizedKeyword = keyword.trim().toLowerCase(Locale.ROOT); return books.values().stream() .filter(book -> book.getTitle().toLowerCase(Locale.ROOT) .contains(normalizedKeyword) || book.getAuthor().toLowerCase(Locale.ROOT) .contains(normalizedKeyword)) .sorted(Comparator.comparing(Book::getTitle)) .collect(Collectors.toList()); } /** * 展示所有图书 */ public void displayAllBooks() { System.out.println("\n========== 图书列表 =========="); books.values().stream() .sorted(Comparator.comparing(Book::getId)) .forEach(book -> System.out.println( "编号：" + book.getId() + "，书名：" + book.getTitle() + "，作者：" + book.getAuthor() + "，库存：" + book.getAvailableCount() + "/" + book.getTotalCount() )); } /** * 查询读者借阅记录 */ public void displayReaderBorrowingRecords(String readerId) { Reader reader = readers.get(readerId); if (reader == null) { System.out.println("读者不存在。"); return; } System.out.println("\n========== " + reader.getName() + " 的借阅记录 =========="); borrowRecords.values().stream() .filter(record -> record.getReaderId().equals(readerId)) .sorted(Comparator.comparing(BorrowRecord::getBorrowDate).reversed()) .forEach(record -> { Book book = books.get(record.getBookId()); System.out.println( "图书：" + (book == null ? "未知图书" : book.getTitle()) + "，借阅日期：" + record.getBorrowDate() + "，应还日期：" + record.getDueDate() + "，状态：" + (record.isReturned() ? "已归还" : "借阅中") ); }); } /** * 统计图书馆数据 */ public void displayStatistics() { int totalBooks = books.values().stream() .mapToInt(Book::getTotalCount) .sum(); int availableBooks = books.values().stream() .mapToInt(Book::getAvailableCount) .sum(); long activeRecords = borrowRecords.values().stream() .filter(record -> !record.isReturned()) .count(); long overdueRecords = borrowRecords.values().stream() .filter(record -> !record.isReturned()) .filter(record -> record.getDueDate().isBefore(LocalDate.now())) .count(); System.out.println("\n========== 图书馆统计 =========="); System.out.println("图书总册数：" + totalBooks); System.out.println("当前可借册数：" + availableBooks); System.out.println("注册读者数：" + readers.size()); System.out.println("当前借阅记录数：" + activeRecords); System.out.println("逾期未还记录数：" + overdueRecords); } static class Book { private final String id; private final String title; private final String author; private final int totalCount; private int availableCount; public Book(String id, String title, String author, int totalCount) { if (totalCount <= 0) { throw new IllegalArgumentException("图书总数量必须大于零"); } this.id = id; this.title = title; this.author = author; this.totalCount = totalCount; this.availableCount = totalCount; } public String getId() { return id; } public String getTitle() { return title; } public String getAuthor() { return author; } public int getTotalCount() { return totalCount; } public int getAvailableCount() { return availableCount; } public void setAvailableCount(int availableCount) { if (availableCount < 0 || availableCount > totalCount) { throw new IllegalArgumentException("库存数量不合法"); } this.availableCount = availableCount; } } static class Reader { private final String id; private final String name; private final String phone; private final Set borrowedBookIds = new HashSet<>(); public Reader(String id, String name, String phone) { this.id = id; this.name = name; this.phone = phone; } public String getId() { return id; } public String getName() { return name; } public String getPhone() { return phone; } public Set getBorrowedBookIds() { return borrowedBookIds; } } static class BorrowRecord { private final String recordId; private final String readerId; private final String bookId; private final LocalDate borrowDate; private final LocalDate dueDate; private LocalDate returnDate; private boolean returned; public BorrowRecord( String recordId, String readerId, String bookId, LocalDate borrowDate, LocalDate dueDate ) { this.recordId = recordId; this.readerId = readerId; this.bookId = bookId; this.borrowDate = borrowDate; this.dueDate = dueDate; this.returned = false; } public LocalDate getBorrowDate() { return borrowDate; } public LocalDate getDueDate() { return dueDate; } public LocalDate getReturnDate() { return returnDate; } public void setReturnDate(LocalDate returnDate) { this.returnDate = returnDate; } public boolean isReturned() { return returned; } public void setReturned(boolean returned) { this.returned = returned; } } }
