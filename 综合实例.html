import java.util.ArrayList; import java.util.List; public class StudentManager { static class Student { private String name; private double score; public Student(String name, double score) { this.name = name; this.score = score; } public String getName() { return name; } public double getScore() { return score; } } public static void main(String[] args) { List students = new ArrayList<>(); students.add(new Student("张三", 88)); students.add(new Student("李四", 92)); students.add(new Student("王五", 76)); double total = 0; System.out.println("===== 学生成绩表 ====="); for (Student student : students) { System.out.printf( "姓名：%s，成绩：%.1f%n", student.getName(), student.getScore() ); total += student.getScore(); } double average = total / students.size(); System.out.printf("班级平均分：%.2f%n", average); } }
