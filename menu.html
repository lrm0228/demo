package main import ( "fmt" ) func calculate(a float64, operator string, b float64) (float64, error) { switch operator { case "+": return a + b, nil case "-": return a - b, nil case "*": return a * b, nil case "/": if b == 0 { return 0, fmt.Errorf("除数不能为零") } return a / b, nil default: return 0, fmt.Errorf("不支持的运算符: %s", operator) } } func main() { var a, b float64 var operator string fmt.Println("=== Go 计算器 ===") fmt.Print("请输入第一个数字: ") if _, err := fmt.Scan(&a); err != nil { fmt.Println("输入错误:", err) return } fmt.Print("请输入运算符 (+ - * /): ") if _, err := fmt.Scan(&operator); err != nil { fmt.Println("输入错误:", err) return } fmt.Print("请输入第二个数字: ") if _, err := fmt.Scan(&b); err != nil { fmt.Println("输入错误:", err) return } result, err := calculate(a, operator, b) if err != nil { fmt.Println("计算失败:", err) return } fmt.Printf("计算结果: %.2f %s %.2f = %.2f\n", a, operator, b, result) }
