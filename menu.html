package main import ( "fmt" "sort" "strings" ) type Student struct { ID string Name string Age int Major string Score float64 } type StudentManager struct { students map[string]Student } func NewStudentManager() *StudentManager { return &StudentManager{ students: make(map[string]Student), } } func (m *StudentManager) AddStudent(s Student) error { if strings.TrimSpace(s.ID) == "" { return fmt.Errorf("学号不能为空") } if strings.TrimSpace(s.Name) == "" { return fmt.Errorf("姓名不能为空") } if s.Age < 1 || s.Age > 120 { return fmt.Errorf("年龄必须在 1 到 120 之间") } if s.Score < 0 || s.Score > 100 { return fmt.Errorf("成绩必须在 0 到 100 之间") } if _, exists := m.students[s.ID]; exists { return fmt.Errorf("学号 %s 已存在", s.ID) } m.students[s.ID] = s return nil } func (m *StudentManager) FindStudent(id string) (Student, bool) { s, exists := m.students[id] return s, exists } func (m *StudentManager) DeleteStudent(id string) error { if _, exists := m.students[id]; !exists { return fmt.Errorf("未找到学号为 %s 的学生", id) } delete(m.students, id) return nil } func (m *StudentManager) ListStudents() []Student { result := make([]Student, 0, len(m.students)) for _, student := range m.students { result = append(result, student) } sort.Slice(result, func(i, j int) bool { return result[i].ID < result[j].ID }) return result } func (m *StudentManager) SearchByName(keyword string) []Student { var result []Student keyword = strings.ToLower(keyword) for _, student := range m.students { if strings.Contains(strings.ToLower(student.Name), keyword) { result = append(result, student) } } return result } func (m *StudentManager) ShowStatistics() { students := m.ListStudents() if len(students) == 0 { fmt.Println("暂无学生数据") return } var total float64 var highest, lowest = students[0].Score, students[0].Score passCount := 0 for _, student := range students { total += student.Score if student.Score > highest { highest = student.Score } if student.Score < lowest { lowest = student.Score } if student.Score >= 60 { passCount++ } } average := total / float64(len(students)) fmt.Printf("学生总数: %d\n", len(students)) fmt.Printf("平均成绩: %.2f\n", average) fmt.Printf("最高成绩: %.2f\n", highest) fmt.Printf("最低成绩: %.2f\n", lowest) fmt.Printf("及格人数: %d\n", passCount) fmt.Printf("及格率: %.2f%%\n", float64(passCount)/float64(len(students))*100) } func main() { manager := NewStudentManager() _ = manager.AddStudent(Student{ ID: "S001", Name: "张三", Age: 20, Major: "计算机科学", Score: 88.5, }) _ = manager.AddStudent(Student{ ID: "S002", Name: "李四", Age: 21, Major: "软件工程", Score: 76.0, }) _ = manager.AddStudent(Student{ ID: "S003", Name: "王五", Age: 19, Major: "人工智能", Score: 92.0, }) for { fmt.Println("\n=== 学生信息管理系统 ===") fmt.Println("1. 查看所有学生") fmt.Println("2. 按学号查询") fmt.Println("3. 添加学生") fmt.Println("4. 删除学生") fmt.Println("5. 按姓名搜索") fmt.Println("6. 查看成绩统计") fmt.Println("0. 退出系统") fmt.Print("请选择功能: ") var choice string if _, err := fmt.Scan(&choice); err != nil { fmt.Println("输入错误:", err) return } switch choice { case "1": students := manager.ListStudents() for _, s := range students { fmt.Printf("学号: %s | 姓名: %s | 年龄: %d | 专业: %s | 成绩: %.2f\n", s.ID, s.Name, s.Age, s.Major, s.Score) } case "2": var id string fmt.Print("请输入学号: ") fmt.Scan(&id) if student, exists := manager.FindStudent(id); exists { fmt.Printf("查询结果: %+v\n", student) } else { fmt.Println("未找到该学生") } case "3": var s Student fmt.Print("学号: ") fmt.Scan(&s.ID) fmt.Print("姓名: ") fmt.Scan(&s.Name) fmt.Print("年龄: ") fmt.Scan(&s.Age) fmt.Print("专业: ") fmt.Scan(&s.Major) fmt.Print("成绩: ") fmt.Scan(&s.Score) if err := manager.AddStudent(s); err != nil { fmt.Println("添加失败:", err) } else { fmt.Println("学生添加成功") } case "4": var id string fmt.Print("请输入要删除的学号: ") fmt.Scan(&id) if err := manager.DeleteStudent(id); err != nil { fmt.Println("删除失败:", err) } else { fmt.Println("删除成功") } case "5": var keyword string fmt.Print("请输入姓名关键词: ") fmt.Scan(&keyword) results := manager.SearchByName(keyword) if len(results) == 0 { fmt.Println("没有找到匹配的学生") continue } for _, s := range results { fmt.Printf("%s - %s - 成绩: %.2f\n", s.ID, s.Name, s.Score) } case "6": manager.ShowStatistics() case "0": fmt.Println("系统已退出") return default: fmt.Println("无效选项，请重新输入") } } }
