import java.util.ArrayList; import java.util.Comparator; import java.util.List; import java.util.Scanner; public class StudentInformationSystem { static class Student { private final String studentId; private String name; private int age; private String major; private double score; public Student(String studentId, String name, int age, String major, double score) { this.studentId = studentId; this.name = name; this.age = age; this.major = major; this.score = score; } public String getStudentId() { return studentId; } public String getName() { return name; } public int getAge() { return age; } public String getMajor() { return major; } public double getScore() { return score; } public void update(String name, int age, String major, double score) { if (name == null || name.trim().isEmpty()) { throw new IllegalArgumentException("姓名不能为空"); } if (age < 1 || age > 120) { throw new IllegalArgumentException("年龄不合法"); } if (score < 0 || score > 100) { throw new IllegalArgumentException("成绩必须在0到100之间"); } this.name = name; this.age = age; this.major = major; this.score = score; } @Override public String toString() { return String.format( "学号：%s，姓名：%s，年龄：%d，专业：%s，成绩：%.2f", studentId, name, age, major, score ); } } static class StudentService { private final List students = new ArrayList<>(); public void addStudent(Student student) { if (findById(student.getStudentId()) != null) { throw new IllegalArgumentException("学号已存在"); } students.add(student); } public Student findById(String studentId) { for (Student student : students) { if (student.getStudentId().equals(studentId)) { return student; } } return null; } public boolean deleteStudent(String studentId) { Student student = findById(studentId); if (student == null) { return false; } students.remove(student); return true; } public boolean updateStudent(String studentId, String name, int age, String major, double score) { Student student = findById(studentId); if (student == null) { return false; } student.update(name, age, major, score); return true; } public void listStudents() { if (students.isEmpty()) { System.out.println("暂无学生信息"); return; } for (Student student : students) { System.out.println(student); } } public void sortByScore() { students.sort( Comparator.comparingDouble(Student::getScore).reversed() ); System.out.println("已按成绩从高到低排序"); listStudents(); } public void searchByName(String keyword) { boolean found = false; for (Student student : students) { if (student.getName().contains(keyword)) { System.out.println(student); found = true; } } if (!found) { System.out.println("未找到符合条件的学生"); } } public void showStatistics() { if (students.isEmpty()) { System.out.println("暂无数据，无法统计"); return; } double total = 0; double highest = Double.MIN_VALUE; double lowest = Double.MAX_VALUE; int passed = 0; for (Student student : students) { double score = student.getScore(); total += score; highest = Math.max(highest, score); lowest = Math.min(lowest, score); if (score >= 60) { passed++; } } System.out.printf("学生总数：%d%n", students.size()); System.out.printf("平均成绩：%.2f%n", total / students.size()); System.out.printf("最高成绩：%.2f%n", highest); System.out.printf("最低成绩：%.2f%n", lowest); System.out.printf("及格人数：%d%n", passed); System.out.printf("及格率：%.2f%%%n", passed * 100.0 / students.size()); } } public static void main(String[] args) { Scanner scanner = new Scanner(System.in); StudentService service = new StudentService(); service.addStudent(new Student("S001", "张三", 20, "计算机科学", 88)); service.addStudent(new Student("S002", "李四", 21, "软件工程", 93)); service.addStudent(new Student("S003", "王五", 19, "人工智能", 57)); while (true) { System.out.println("\n===== 学生信息管理系统 ====="); System.out.println("1. 添加学生"); System.out.println("2. 查询所有学生"); System.out.println("3. 根据学号查询"); System.out.println("4. 修改学生信息"); System.out.println("5. 删除学生"); System.out.println("6. 按姓名搜索"); System.out.println("7. 按成绩排序"); System.out.println("8. 成绩统计"); System.out.println("0. 退出"); System.out.print("请输入选项："); String choice = scanner.nextLine().trim(); try { switch (choice) { case "1": System.out.print("学号："); String id = scanner.nextLine().trim(); System.out.print("姓名："); String name = scanner.nextLine().trim(); System.out.print("年龄："); int age = Integer.parseInt(scanner.nextLine().trim()); System.out.print("专业："); String major = scanner.nextLine().trim(); System.out.print("成绩："); double score = Double.parseDouble( scanner.nextLine().trim()); Student student = new Student( id, name, age, major, score); student.update(name, age, major, score); service.addStudent(student); System.out.println("添加成功"); break; case "2": service.listStudents(); break; case "3": System.out.print("请输入学号："); Student found = service.findById( scanner.nextLine().trim()); System.out.println(found == null ? "学生不存在" : found); break; case "4": System.out.print("学号："); String updateId = scanner.nextLine().trim(); System.out.print("新姓名："); String newName = scanner.nextLine().trim(); System.out.print("新年龄："); int newAge = Integer.parseInt( scanner.nextLine().trim()); System.out.print("新专业："); String newMajor = scanner.nextLine().trim(); System.out.print("新成绩："); double newScore = Double.parseDouble( scanner.nextLine().trim()); if (service.updateStudent(updateId, newName, newAge, newMajor, newScore)) { System.out.println("修改成功"); } else { System.out.println("学生不存在"); } break; case "5": System.out.print("请输入学号："); boolean deleted = service.deleteStudent( scanner.nextLine().trim()); System.out.println(deleted ? "删除成功" : "学生不存在"); break; case "6": System.out.print("姓名关键词："); service.searchByName(scanner.nextLine().trim()); break; case "7": service.sortByScore(); break; case "8": service.showStatistics(); break; case "0": scanner.close(); System.out.println("系统已退出"); return; default: System.out.println("无效选项"); } } catch (IllegalArgumentException e) { System.out.println("输入或操作错误：" + e.getMessage()); } } } }
